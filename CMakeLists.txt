cmake_minimum_required(VERSION 3.16)
project(cpp_hll LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(test_hll tests/test_hll.cpp)
target_include_directories(test_hll PRIVATE include)
if(MSVC)
  target_compile_options(test_hll PRIVATE /W4)
else()
  target_compile_options(test_hll PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME test_hll COMMAND test_hll)
